display : flex;
unlocks = justify-content & align items
align items = y axis
justify-content = x-axis

flex-direction = column or row ;
flex-direction = column.reverse-column,row,reverse-row

flex - wrap : wrap & nowrap ;
wrap = keep only number of boxes which can fit in a line 
       other will be moved to next line
no wrap = keep all the boxes in one line but if number of 
          boxes increase it will increase the width the line will be broder


flex-shrink = shrinks the boxes less means no shrink high means more shrink
